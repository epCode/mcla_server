[Bedrock portals]

#    Nether portals teleport to a chamber just above bedrock in the Overworld
#    (and back) instead of leading to the Nether.
mcla_server_bedrock_portals (Bedrock portals instead of the Nether) bool true

#    Portals at or below this height send you back up to the surface; portals
#    above it send you down to bedrock level. The default is 16 nodes above the
#    top of the bedrock layer.
mcla_server_deep_portal_threshold (Deep portal Y threshold) int -108

[Overworld ores]

#    Generate veins of quartz ore deep underground in the Overworld.
mcla_server_overworld_quartz (Overworld quartz ore) bool true

#    Generate ancient debris in the Overworld, level with the deep dark.
mcla_server_overworld_ancient_debris (Overworld ancient debris) bool true

#    Quartz veins per chunk. Singlenode (mcl_levelgen) worlds only.
mcla_server_quartz_veins_per_chunk (Quartz veins per chunk) int 3 0 32

#    One in this many chunks gets an ancient debris vein. Singlenode
#    (mcl_levelgen) worlds only. Higher is rarer.
mcla_server_ancient_debris_rarity (Ancient debris rarity) int 4 1 256

#    Nodes per quartz vein attempt. Builtin mapgens (v7 and friends) only.
#    Higher is rarer.
mcla_server_quartz_scarcity (Quartz ore scarcity) int 3400 100 1000000

#    Nodes per ancient debris vein attempt. Builtin mapgens (v7 and friends)
#    only. Higher is rarer.
mcla_server_ancient_debris_scarcity (Ancient debris scarcity) int 26000 1000 10000000

[Netherite]

#    Make the netherite upgrade template craftable, since its only vanilla
#    source is a bastion remnant chest in the Nether.
mcla_server_netherite_recipes (Overworld netherite upgrade template) bool true

[Blaze spawners]

#    Give dungeon mob spawners a chance of being blaze spawners, so blaze rods
#    and brewing stay reachable without the Nether.
mcla_server_blaze_spawners (Blaze dungeon spawners) bool true

#    One in this many dungeon spawners is a blaze spawner. The remaining seven
#    eighths keep vanilla's zombie / spider / skeleton split, which makes a
#    blaze spawner about twice as rare as any of them.
mcla_server_blaze_spawner_chance (Blaze spawner rarity) int 8 1 256
